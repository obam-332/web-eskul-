#include <WiFi.h>
#include <WebServer.h>

const char* ssid = "NAMA_WIFI_ANDA";
const char* password = "PASSWORD_WIFI_ANDA";

const int RELAY_PIN = 23; // Pin Relay (Active LOW / HIGH sesuaikan)
bool relayState = false;

// Variables for Timers
unsigned long timerTargetTime = 0;
bool isTimerActive = false;
bool timerTargetState = false; // false = OFF, true = ON

WebServer server(80);

void handleRoot() {
  server.send(200, "text/html", "ESP32 Ready");
}

void handleStatus() {
  String json = "{\"status\":\"" + String(relayState ? "ON" : "OFF") + "\",";
  
  long remainingSec = 0;
  if(isTimerActive) {
    long remainingMillis = timerTargetTime - millis();
    if(remainingMillis > 0) {
      remainingSec = remainingMillis / 1000;
    }
  }
  
  json += "\"timerActive\":" + String(isTimerActive ? "true" : "false") + ",";
  json += "\"timerType\":\"" + String(timerTargetState ? "ON" : "OFF") + "\",";
  json += "\"remainingSec\":" + String(remainingSec) + "}";
  
  server.sendHeader("Access-Control-Allow-Origin", "*");
  server.send(200, "application/json", json);
}

void handleOn() {
  relayState = true;
  digitalWrite(RELAY_PIN, LOW); // LOW = Relay ON (Kebanyakan module relay Active LOW)
  isTimerActive = false; // Reset timer jika di-override manual
  server.sendHeader("Access-Control-Allow-Origin", "*");
  server.send(200, "text/plain", "OK_ON");
}

void handleOff() {
  relayState = false;
  digitalWrite(RELAY_PIN, HIGH); // HIGH = Relay OFF
  isTimerActive = false; // Reset timer jika di-override manual
  server.sendHeader("Access-Control-Allow-Origin", "*");
  server.send(200, "text/plain", "OK_OFF");
}

void handleTimerOff() {
  if (server.hasArg("sec")) {
    int sec = server.arg("sec").toInt();
    if (sec > 0) {
      timerTargetTime = millis() + ((unsigned long)sec * 1000);
      isTimerActive = true;
      timerTargetState = false; // Goal: TURN OFF
    }
  }
  server.sendHeader("Access-Control-Allow-Origin", "*");
  server.send(200, "text/plain", "TIMER_OFF_SET");
}

void handleTimerOn() {
  if (server.hasArg("sec")) {
    int sec = server.arg("sec").toInt();
    if (sec > 0) {
      timerTargetTime = millis() + ((unsigned long)sec * 1000);
      isTimerActive = true;
      timerTargetState = true; // Goal: TURN ON
    }
  }
  server.sendHeader("Access-Control-Allow-Origin", "*");
  server.send(200, "text/plain", "TIMER_ON_SET");
}

void handleCancelTimer() {
  isTimerActive = false;
  server.sendHeader("Access-Control-Allow-Origin", "*");
  server.send(200, "text/plain", "TIMER_CANCELLED");
}

void setup() {
  Serial.begin(115200);
  pinMode(RELAY_PIN, OUTPUT);
  digitalWrite(RELAY_PIN, HIGH); // Initial Off

  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("");
  Serial.print("IP Address ESP32: ");
  Serial.println(WiFi.localIP());

  server.on("/", handleRoot);
  server.on("/status", handleStatus);
  server.on("/on", handleOn);
  server.on("/off", handleOff);
  server.on("/timer-off", handleTimerOff);
  server.on("/timer-on", handleTimerOn);
  server.on("/cancel-timer", handleCancelTimer);

  server.begin();
}

void loop() {
  server.handleClient();

  // Check Timer Execution
  if (isTimerActive && millis() >= timerTargetTime) {
    isTimerActive = false;
    if (timerTargetState == true) {
      relayState = true;
      digitalWrite(RELAY_PIN, LOW); // Turn ON
    } else {
      relayState = false;
      digitalWrite(RELAY_PIN, HIGH); // Turn OFF
    }
  }
}
